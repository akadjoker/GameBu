project(builder)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

file(GLOB_RECURSE BUILDER_SOURCES CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp"
)

add_executable(builder ${BUILDER_SOURCES})

target_include_directories(builder PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/include
    ${CMAKE_CURRENT_SOURCE_DIR}/third_party
)

find_package(Threads REQUIRED)
target_link_libraries(builder PRIVATE Threads::Threads)

if (WIN32)
    target_link_libraries(builder PRIVATE ws2_32)
endif()

set(BU_BUILDER_OUTPUT_DIR "${CMAKE_SOURCE_DIR}/bin")
set_target_properties(builder PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${BU_BUILDER_OUTPUT_DIR}"
    RUNTIME_OUTPUT_DIRECTORY_DEBUG "${BU_BUILDER_OUTPUT_DIR}"
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${BU_BUILDER_OUTPUT_DIR}"
    RUNTIME_OUTPUT_DIRECTORY_RELWITHDEBINFO "${BU_BUILDER_OUTPUT_DIR}"
    RUNTIME_OUTPUT_DIRECTORY_MINSIZEREL "${BU_BUILDER_OUTPUT_DIR}"
)
